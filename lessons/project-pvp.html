Проект: PvP-оружие

Делаем бластер для PvP: игрок целится мышью, сервер проверяет выстрел и считает попадание, все видят трассер, а за убийства начисляются очки. Главная задача проекта — сделать оружие, которое нельзя взломать простыми читами.

🎯 Что получится:
  • Tool с LocalScript внутри: прицеливание и отправка выстрела;
  • серверная проверка: кулдаун, дистанция, луч (raycast) от позиции, которую считает сервер;
  • урон, хедшоты, учёт убийств и смертей;
  • визуальные эффекты на клиентах: трассеры и хитмаркер.

Как разделена работа

Клиент (LocalScript)Сервер (Script)
Узнаёт, куда смотрит курсорПроверяет, что оружие в руках и кулдаун прошёл
Шлёт «стреляю в эту точку»Сам считает точку вылета пули и пускает луч
Рисует трассеры и хитмаркерРешает, есть ли попадание, наносит урон, считает убийства

Чего клиент не присылает: кого он поразил, сколько урона, откуда летит пуля. Всё это читер подделал бы первым делом.

Шаг 1. Структура

Explorer
  • ReplicatedStorage
    • Shared
      • WeaponConfig
    • Remotes
      • Fire RemoteEvent: клиент → сервер
      • ShotEffect RemoteEvent: сервер → все
      • HitMarker RemoteEvent: сервер → стрелок
  • ServerScriptService
    • CombatServer
  • StarterPack
    • Blaster Tool
      • Handle Part 1×1×3, CanCollide = false
      • BlasterClient
  • StarterPlayer
    • StarterPlayerScripts
      • EffectsClient
  • Tool в StarterPack копируется в рюкзак (Backpack) игрока при каждом появлении.
  • Деталь внутри Tool обязательно должна называться Handle: за неё персонаж держит оружие.
  • LocalScript внутри Tool работает, потому что Tool лежит в Backpack игрока или в его персонаже. Это ещё одно законное место для LocalScript.

Шаг 2. Настройки оружия

local WeaponConfig = {}

WeaponConfig.Blaster = {
	damage = 20,
	headshotMultiplier = 1.5,
	cooldown = 0.35,       -- секунд между выстрелами
	range = 300,           -- дальность в студах
}

return WeaponConfig

Шаг 3. Клиент: прицеливание и выстрел

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UserInputService = game:GetService("UserInputService")

local tool = script.Parent
local player = Players.LocalPlayer
local camera = workspace.CurrentCamera

local config = require(ReplicatedStorage:WaitForChild("Shared"):WaitForChild("WeaponConfig")).Blaster
local fireRemote = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("Fire")

local lastShot = 0

-- Точка в мире под курсором (на телефоне — место касания)
local function getAimPosition()
	local screenPos = UserInputService:GetMouseLocation()
	local ray = camera:ViewportPointToRay(screenPos.X, screenPos.Y)

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {player.Character}  -- не целиться в себя

	local result = workspace:Raycast(ray.Origin, ray.Direction * 1000, params)
	if result then
		return result.Position
	end
	return ray.Origin + ray.Direction * 1000
end

tool.Activated:Connect(function()
	-- Клиентский кулдаун только для отзывчивости. Настоящий — на сервере
	local now = os.clock()
	if now - lastShot < config.cooldown then return end
	lastShot = now

	fireRemote:FireServer(getAimPosition())
end)

Клиент тоже проверяет кулдаун, чтобы не спамить сервер запросами. Но доверять этой проверке нельзя: читер её просто удалит. Поэтому сервер проверяет кулдаун ещё раз.

Шаг 4. Сервер: проверка и урон

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Debris = game:GetService("Debris")

local config = require(ReplicatedStorage.Shared.WeaponConfig).Blaster
local remotes = ReplicatedStorage.Remotes
local fireRemote = remotes.Fire
local shotEffect = remotes.ShotEffect
local hitMarker = remotes.HitMarker

local lastShot = {}  -- [player] = время последнего выстрела

-- ===== Проверки =====
local function isValidVector(v)
	if typeof(v) ~= "Vector3" then return false end
	-- NaN не равен сам себе; inf тоже отсекаем
	for _, n in {v.X, v.Y, v.Z} do
		if n ~= n or math.abs(n) == math.huge then
			return false
		end
	end
	return true
end

local function getEquippedBlaster(character)
	local tool = character:FindFirstChild("Blaster")  -- экипированный Tool лежит в персонаже
	if tool and tool:IsA("Tool") then
		return tool
	end
	return nil
end

local function areTeammates(a, b)
	return a.Team ~= nil and a.Team == b.Team and not a.Neutral
end

-- ===== Выстрел =====
fireRemote.OnServerEvent:Connect(function(player, aimPosition)
	-- 1. Данные от клиента правильного типа?
	if not isValidVector(aimPosition) then return end

	-- 2. Стрелок жив и держит оружие?
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid or humanoid.Health <= 0 then return end
	local tool = getEquippedBlaster(character)
	if not tool then return end
	local handle = tool:FindFirstChild("Handle")
	if not handle then return end

	-- 3. Кулдаун (на сервере, с небольшим допуском на пинг)
	local now = os.clock()
	if lastShot[player] and now - lastShot[player] < config.cooldown * 0.9 then return end
	lastShot[player] = now

	-- 4. Точку вылета считает СЕРВЕР, а не клиент
	local origin = handle.Position
	local direction = aimPosition - origin
	if direction.Magnitude < 0.1 then return end
	direction = direction.Unit * config.range

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {character}

	local result = workspace:Raycast(origin, direction, params)
	local hitPosition = if result then result.Position else origin + direction

	-- 5. Эффект выстрела видят все
	shotEffect:FireAllClients(origin, hitPosition)

	if not result then return end

	-- 6. Попали в персонажа?
	local model = result.Instance:FindFirstAncestorOfClass("Model")
	local targetHumanoid = model and model:FindFirstChildOfClass("Humanoid")
	if not targetHumanoid or targetHumanoid.Health <= 0 then return end

	local targetPlayer = Players:GetPlayerFromCharacter(model)
	if targetPlayer and areTeammates(player, targetPlayer) then return end

	-- 7. Урон
	local isHeadshot = result.Instance.Name == "Head"
	local damage = config.damage * (if isHeadshot then config.headshotMultiplier else 1)

	-- Метка «кто ударил последним», чтобы засчитать убийство
	local oldTag = targetHumanoid:FindFirstChild("creator")
	if oldTag then oldTag:Destroy() end
	local tag = Instance.new("ObjectValue")
	tag.Name = "creator"
	tag.Value = player
	tag.Parent = targetHumanoid
	Debris:AddItem(tag, 5)

	targetHumanoid:TakeDamage(damage)  -- учитывает ForceField после спавна
	hitMarker:FireClient(player, isHeadshot)
end)

-- ===== Статистика: убийства и смерти =====
local function onCharacterAdded(player, character)
	local humanoid = character:WaitForChild("Humanoid")
	humanoid.Died:Connect(function()
		player.leaderstats.Deaths.Value += 1

		local tag = humanoid:FindFirstChild("creator")
		local killer = tag and tag.Value
		if killer and killer ~= player and killer.Parent then
			killer.leaderstats.Kills.Value += 1
		end
	end)
end

Players.PlayerAdded:Connect(function(player)
	local leaderstats = Instance.new("Folder")
	leaderstats.Name = "leaderstats"

	local kills = Instance.new("IntValue")
	kills.Name = "Kills"
	kills.Parent = leaderstats

	local deaths = Instance.new("IntValue")
	deaths.Name = "Deaths"
	deaths.Parent = leaderstats

	leaderstats.Parent = player

	player.CharacterAdded:Connect(function(character)
		onCharacterAdded(player, character)
	end)
end)

Players.PlayerRemoving:Connect(function(player)
	lastShot[player] = nil
end)

Разбор защиты

ЧитЧто его останавливает
Стрельба без оружия в рукахПроверка getEquippedBlaster
Пулемёт: 100 выстрелов в секундуСерверный кулдаун lastShot
Стрельба через всю карту или сквозь стеныЛуч пускает сервер из позиции оружия, с ограничением дальности. Стена его остановит
«Я попал в игрока X на 9999 урона»Клиент не сообщает ни цель, ни урон, только точку прицела
Мусор в аргументах (строка, NaN, inf)isValidVector. Без проверки NaN мог бы сломать расчёты
Стрельба мёртвым персонажемПроверка humanoid.Health

Что эта система не ловит: аимбот (читер идеально наводит курсор). От него защищаются статистикой: слишком высокий процент хедшотов, мгновенные повороты. Это уже продвинутая тема. Абсолютной защиты нет, но базовые проверки отсекают 95% читов.

Шаг 5. Эффекты на клиентах

Трассер рисует каждый клиент у себя: так сервер не тратит ресурсы на визуал, а эффект появляется плавно.

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TweenService = game:GetService("TweenService")
local Debris = game:GetService("Debris")

local player = Players.LocalPlayer
local remotes = ReplicatedStorage:WaitForChild("Remotes")
local shotEffect = remotes:WaitForChild("ShotEffect")
local hitMarker = remotes:WaitForChild("HitMarker")

-- ===== Трассер =====
local fadeInfo = TweenInfo.new(0.15)

shotEffect.OnClientEvent:Connect(function(origin, hitPosition)
	local distance = (hitPosition - origin).Magnitude
	local beam = Instance.new("Part")
	beam.Anchored = true
	beam.CanCollide = false
	beam.CanQuery = false       -- лучи не будут в него попадать
	beam.Material = Enum.Material.Neon
	beam.Color = Color3.fromRGB(255, 80, 80)
	beam.Size = Vector3.new(0.15, 0.15, distance)
	-- ставим посередине и поворачиваем от origin к цели
	beam.CFrame = CFrame.lookAt(origin, hitPosition) * CFrame.new(0, 0, -distance / 2)
	beam.Parent = workspace

	TweenService:Create(beam, fadeInfo, {Transparency = 1}):Play()
	Debris:AddItem(beam, 0.2)
end)

-- ===== Хитмаркер =====
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "HitMarkerGui"
screenGui.ResetOnSpawn = false
screenGui.Parent = player:WaitForChild("PlayerGui")

local marker = Instance.new("TextLabel")
marker.AnchorPoint = Vector2.new(0.5, 0.5)
marker.Position = UDim2.fromScale(0.5, 0.5)
marker.Size = UDim2.fromOffset(60, 60)
marker.BackgroundTransparency = 1
marker.Text = "✖"
marker.TextScaled = true
marker.TextTransparency = 1
marker.Parent = screenGui

hitMarker.OnClientEvent:Connect(function(isHeadshot)
	marker.TextColor3 = if isHeadshot then Color3.fromRGB(255, 70, 70) else Color3.new(1, 1, 1)
	marker.TextTransparency = 0
	TweenService:Create(marker, TweenInfo.new(0.3), {TextTransparency = 1}):Play()
end)

Хитмаркер приходит от сервера только при настоящем попадании, поэтому игрок видит честный результат, а не догадку клиента.

Шаг 6. Проверка

  1. Test → Clients and Servers → 2 игрока.
  2. У обоих в рюкзаке бластер. Возьми его в руку (1) и стреляй кликом.
  3. Трассер видят оба игрока. При попадании появляется хитмаркер, у цели уменьшается здоровье.
  4. Хедшот — красный хитмаркер и больше урона. После убийства в таблице +1 Kills и +1 Deaths.
  5. Сразу после возрождения игрок под силовым полем (ForceField от SpawnLocation) и урона не получает.
  6. Стреляй через стену: попадания нет.

🧠 Проверь себя

Почему точку вылета пули берёт сервер (handle.Position), а не клиент?

  1. Так быстрее
  2. Иначе читер прислал бы точку вылета прямо рядом с целью или за стеной
  3. Клиент не знает позицию оружия
  4. Так требует Raycast

Всё, что влияет на результат, сервер считает сам.

Почему трассеры рисует клиент, а не сервер?

  1. Сервер не умеет создавать детали
  2. Клиент сам знает, кто попал
  3. Так безопаснее
  4. Визуальные эффекты дешевле и плавнее рисовать на клиентах, сервер только сообщает о выстреле

Сервер занят логикой, клиенты рисуют.

Зачем на Humanoid жертвы вешается ObjectValue creator?

  1. Чтобы при смерти знать, кто нанёс последний удар, и засчитать убийство
  2. Чтобы нанести урон
  3. Для хитмаркера
  4. Чтобы жертва не возродилась

Это классическая схема Roblox для учёта убийств.

Почему LocalScript внутри Tool работает?

  1. Tool — особый скрипт
  2. Его запускает сервер
  3. Tool находится в Backpack или персонаже игрока, а это места, где LocalScript работает
  4. Он не работает, нужен StarterPlayerScripts

Backpack и персонаж принадлежат игроку.

local SHOT_SOUND_ID = "rbxassetid://ВАШ_ID"

local function playShotSound(position)

Задание 2 среднее

Сделай меч ближнего боя. Клиент при Activated просто шлёт Swing без аргументов. Сервер с кулдауном 0.6 с находит врагов в радиусе 6 студов перед персонажем и наносит 30 урона.

💡 Подсказка

Используй workspace:GetPartBoundsInBox(cframe, size, params), где cframe = root.CFrame * CFrame.new(0, 0, -3), size = Vector3.new(5, 5, 6). Собери уникальные Humanoid из найденных деталей: один враг — один удар.

✅ Решение (сервер) 
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local swing = ReplicatedStorage.Remotes.Swing

local DAMAGE = 30
local COOLDOWN = 0.6
local lastSwing = {}

swing.OnServerEvent:Connect(function(player)
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not root or not humanoid or humanoid.Health <= 0 then return end
	if not character:FindFirstChild("Sword") then return end

	local now = os.clock()
	if lastSwing[player] and now - lastSwing[player] < COOLDOWN then return end
	lastSwing[player] = now

	local params = OverlapParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {character}

	local parts = workspace:GetPartBoundsInBox(root.CFrame * CFrame.new(0, 0, -3), Vector3.new(5, 5, 6), params)
	local alreadyHit = {}
	for _, part in parts do
		local model = part:FindFirstAncestorOfClass("Model")
		local target = model and model:FindFirstChildOfClass("Humanoid")
		if target and not alreadyHit[target] and target.Health > 0 then
			alreadyHit[target] = true
			target:TakeDamage(DAMAGE)
		end
	end
end)

Players.PlayerRemoving:Connect(function(player)
	lastSwing[player] = nil
end)

Клиент: Tool Sword с Handle и LocalScript, который на tool.Activated вызывает Swing:FireServer() и проигрывает анимацию удара.

Задание 3 сложное

Добавь патроны: 12 в обойме, перезарядка на R за 1.5 секунды. Количество патронов хранит сервер (атрибут Ammo на Tool, чтобы клиент мог показать его в интерфейсе). Во время перезарядки стрелять нельзя.

💡 Подсказка
  • В WeaponConfig: magazine = 12, reloadTime = 1.5.
  • Новый RemoteEvent Reload. Сервер ставит атрибут Reloading = true, ждёт reloadTime, ставит Ammo = magazine.
  • В обработчике Fire: если tool:GetAttribute("Reloading") или Ammo <= 0, выходим. Иначе Ammo -= 1.
  • Клиент показывает Ammo через tool:GetAttributeChangedSignal("Ammo").
✅ Решение (серверная часть) 
local reloadRemote = remotes.Reload

-- при выдаче оружия (например, в CharacterAdded) выставь начальные патроны:
-- tool:SetAttribute("Ammo", config.magazine)

-- в fireRemote.OnServerEvent после получения tool, ДО кулдауна:
-- if tool:GetAttribute("Reloading") then return end
-- local ammo = tool:GetAttribute("Ammo") or 0
-- if ammo <= 0 then return end
-- tool:SetAttribute("Ammo", ammo - 1)

reloadRemote.OnServerEvent:Connect(function(player)
	local character = player.Character
	local tool = character and getEquippedBlaster(character)
	if not tool or tool:GetAttribute("Reloading") then return end
	if (tool:GetAttribute("Ammo") or 0) >= config.magazine then return end

	tool:SetAttribute("Reloading", true)
	task.wait(config.reloadTime)
	if tool.Parent then
		tool:SetAttribute("Ammo", config.magazine)
		tool:SetAttribute("Reloading", false)
	end
end)

Начальные патроны удобно задать прямо в Studio: атрибут Ammo = 12 на Tool в StarterPack. Он скопируется вместе с инструментом.
